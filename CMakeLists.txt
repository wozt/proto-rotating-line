cmake_minimum_required(VERSION 3.21)
project(proto_rotating_line VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_library(proto_curve STATIC src/curve.c src/config.c src/expression.c src/parametric.c)
target_include_directories(proto_curve PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(proto_curve PUBLIC m)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(proto_curve PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(PROTO_BUILD_APP "Build the SDL3 viewer" ON)
if(PROTO_BUILD_APP)
    find_package(SDL3 CONFIG REQUIRED)
    add_executable(proto_rotating_line src/main.c src/pulse_audio.c)
    target_link_libraries(proto_rotating_line PRIVATE proto_curve SDL3::SDL3)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(proto_rotating_line PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()

include(CTest)

    add_executable(test_curve tests/test_curve.c)
    target_link_libraries(test_curve PRIVATE proto_curve)
    add_test(NAME curve_math COMMAND test_curve)

    target_link_libraries(test_config PRIVATE proto_curve)
    target_compile_definitions(test_config PRIVATE PROTO_SAMPLE_CFG="${CMAKE_CURRENT_SOURCE_DIR}/config/prototype.json")
    target_compile_definitions(test_config PRIVATE PROTO_WAVECREST_CFG="${CMAKE_CURRENT_SOURCE_DIR}/config/prototype_windwave_4x.json")
    target_compile_definitions(test_config PRIVATE PROTO_FILIFORM_CFG="${CMAKE_CURRENT_SOURCE_DIR}/config/prototype_filiforme_4x.json")
    target_compile_definitions(test_config PRIVATE PROTO_SPATIAL_CFG="${CMAKE_CURRENT_SOURCE_DIR}/config/prototype_spatial_gradient_4x.json")
    target_compile_definitions(test_config PRIVATE PROTO_SPATIAL_GROWTH_0013="${CMAKE_CURRENT_SOURCE_DIR}/config/prototype_spatial_growth_0013.json")
    target_compile_definitions(test_config PRIVATE PROTO_RIBBON_CFG="${CMAKE_CURRENT_SOURCE_DIR}/config/prototype_wolfram_ribbon_4x.json")
