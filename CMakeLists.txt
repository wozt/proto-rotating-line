cmake_minimum_required(VERSION 3.21)
project(proto_rotating_line VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_library(proto_curve STATIC src/curve.c src/config.c src/expression.c src/parametric.c)
target_include_directories(proto_curve PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(proto_curve PUBLIC m)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(proto_curve PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(PROTO_BUILD_APP "Build the SDL3 viewer" ON)
if(PROTO_BUILD_APP)
    find_package(SDL3 CONFIG REQUIRED)
    add_executable(proto_rotating_line src/main.c)
    target_link_libraries(proto_rotating_line PRIVATE proto_curve SDL3::SDL3)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(proto_rotating_line PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(test_curve tests/test_curve.c)
    target_link_libraries(test_curve PRIVATE proto_curve)
    add_test(NAME curve_math COMMAND test_curve)
    add_executable(test_config tests/test_config.c)
    target_link_libraries(test_config PRIVATE proto_curve)
    target_compile_definitions(test_config PRIVATE PROTO_SAMPLE_CFG="${CMAKE_CURRENT_SOURCE_DIR}/config/prototype.json")
    add_test(NAME config_roundtrip COMMAND test_config)
endif()
